library(shiny)
library(shinydashboard)
library(tidyverse)
library(plotly)

set.seed(456)
n_cases <- 1200
pv_data <- tibble(
  Case_ID = paste0("EU-2026-", 1000:(1000 + n_cases - 1)),
  Drug_Name = sample(c("Adalimumab", "Metformin", "Ibuprofen", "Rivastigmine"), n_cases, replace = TRUE),
  Event_MedDRA = sample(c("Dizziness", "Nausea", "Acute Kidney Injury", "Hepatic Failure", "Rash", "Anaphylaxis"), n_cases, replace = TRUE, prob = c(0.3, 0.25, 0.15, 0.1, 0.1, 0.1)),
  Seriousness = sample(c("Serious", "Non-Serious"), n_cases, replace = TRUE, prob = c(0.35, 0.65)),
  Submission_Status = sample(c("On-time", "Late"), n_cases, replace = TRUE, prob = c(0.94, 0.06))
)

ui <- dashboardPage(
  skin = "blue",
  dashboardHeader(title = "PV Global Operations"),
  dashboardSidebar(
    sidebarMenu(
      menuItem("Safety Overview", tabName = "dashboard", icon = icon("dashboard")),
      selectInput("drug_filter", "Select Molecule:", choices = unique(pv_data$Drug_Name), selected = "Adalimumab")
    )
  ),
  dashboardBody(
    tags$head(tags$style(HTML(".small-box { background-color: #2c3e50 !important; }"))),
    fluidRow(
      valueBoxOutput("total_cases", width = 4),
      valueBoxOutput("seriousness_rate", width = 4),
      valueBoxOutput("compliance_rate", width = 4)
    ),
    fluidRow(
      box(title = "Adverse Event Distribution (MedDRA Preferred Terms)", status = "primary", solidHeader = TRUE, width = 12,
          plotlyOutput("ae_plot", height = "400px"))
    )
  )
)

server <- function(input, output) {
  filtered_data <- reactive({
    pv_data %>% filter(Drug_Name == input$drug_filter)
  })
  
  output$total_cases <- renderValueBox({
    valueBox(nrow(filtered_data()), "Total Evaluated Cases", icon = icon("file-medical"), color = "navy")
  })
  
  output$seriousness_rate <- renderValueBox({
    pct <- round(mean(filtered_data()$Seriousness == "Serious") * 100, 1)
    valueBox(paste0(pct, "%"), "Serious Adverse Events (SAE)", icon = icon("exclamation-triangle"), color = "orange")
  })
  
  output$compliance_rate <- renderValueBox({
    pct_ontime <- round(mean(filtered_data()$Submission_Status == "On-time") * 100, 1)
    status_color = if(pct_ontime >= 95) "green" else "red"
    valueBox(paste0(pct_ontime, "%"), "On-Time Regulatory Submission", icon = icon("clock"), color = status_color)
  })
  
  output$ae_plot <- renderPlotly({
    p <- filtered_data() %>%
      count(Event_MedDRA) %>%
      ggplot(aes(x = reorder(Event_MedDRA, n), y = n, text = paste("Cases:", n))) +
      geom_bar(stat = "identity", fill = "#3c8dbc") +
      coord_flip() +
      theme_minimal() +
      labs(x = NULL, y = "Report Count")
    
    ggplotly(p, tooltip = "text")
  })
}

shinyApp(ui, server)
